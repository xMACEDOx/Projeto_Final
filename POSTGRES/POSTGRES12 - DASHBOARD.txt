POSTGRES 0.7 - DASHBOARD

CREATE OR REPLACE VIEW public.vw_kpi_dashboard AS
SELECT
    p.id_linha,
    COUNT(*) AS total_registros,
    COUNT(DISTINCT p.prefixo_veiculo) AS frota_ativa,

    --  Atraso médio corrigido (hora_onibus_pos ajustada para timestamp)
    ROUND(
        AVG(
            EXTRACT(
                EPOCH FROM (
                    -- diferença entre hora_onibus_pos ajustada e hora_prevista
                    (
                        to_timestamp(
                            to_char(p.hora_prevista::timestamptz, 'YYYY-MM-DD') || ' ' || p.hora_onibus_pos,
                            'YYYY-MM-DD HH24:MI'
                        ) - p.hora_prevista::timestamptz
                    )
                )
            ) / 60.0
        )::numeric,
        2
    ) AS atraso_medio_min,

    --  Percentual de acessibilidade
    ROUND(
        (SUM(CASE WHEN p.acessivel THEN 1 ELSE 0 END)::numeric / NULLIF(COUNT(*),0)) * 100,
        2
    ) AS acessibilidade_perc,

    --  Percentual de operação ativa
    ROUND(
        (SUM(CASE WHEN LOWER(p.operando_onibus) IN ('s','sim','true','1') THEN 1 ELSE 0 END)::numeric / NULLIF(COUNT(*),0)) * 100,
        2
    ) AS operacao_ativa_perc

FROM public.previsao_posicoes_enriquecida p
WHERE p.hora_onibus_pos IS NOT NULL
  AND p.hora_prevista IS NOT NULL
GROUP BY p.id_linha;
