CREATE TABLE IF NOT EXISTS public.previsao_posicoes_enriquecida (
    id SERIAL PRIMARY KEY,
    timestamp TIMESTAMP,
    hora_coleta TIMESTAMP,
    codigo_parada VARCHAR(50),
    nome_parada VARCHAR(255),
    latitude DOUBLE PRECISION,
    longitude DOUBLE PRECISION,
    prefixo_veiculo VARCHAR(50),
    hora_prevista TIMESTAMP,
    acessivel BOOLEAN,
    id_linha VARCHAR(50),
    route_id VARCHAR(50),
    lat_onibus DOUBLE PRECISION,
    lng_onibus DOUBLE PRECISION,
    hora_onibus_pos TIMESTAMP,
    operando_onibus VARCHAR(10),
    stop_name_gtfs VARCHAR(255),
    stop_lat DOUBLE PRECISION,
    stop_lon DOUBLE PRECISION,
    terminal_origem VARCHAR(255),
    terminal_destino VARCHAR(255)
);
